---
import Base from "../layouts/Base.astro";
import solidaryContent from "../content/solidary.md";
import { parseSolidaryFrontmatter } from "../solidary-config/site";
import { getEntry } from "astro:content";

type MarkdownModule = { frontmatter?: Record<string, unknown> };

const solidary = parseSolidaryFrontmatter((solidaryContent as MarkdownModule).frontmatter);
const siteName = solidary.title || "Solidary Site";
const siteDescription = solidary.description || "Solidary Site Description.";
const siteUrl = solidary.url || "https://example.com";

const jsonLd = {
  "@context": "https://schema.org",
  "@type": "WebSite",
  name: siteName,
  url: siteUrl,
  inLanguage: "en"
};

const homeEntry = await getEntry("pages", "home");
const { Content: HomeContent } = homeEntry ? await homeEntry.render() : { Content: null };
---
<Base title={siteName} description={siteDescription} canonicalPath="/" jsonLd={jsonLd}>
  <article class="prose">
    {HomeContent ? <HomeContent /> : null}
  </article>
</Base>
