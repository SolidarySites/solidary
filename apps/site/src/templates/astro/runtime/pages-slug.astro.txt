---
import Base from "../layouts/Base.astro";
import solidaryContent from "../content/solidary.md";
import { parseSolidaryFrontmatter } from "../solidary-config/site";
import { getCollection } from "astro:content";

type MarkdownModule = { frontmatter?: Record<string, unknown> };

const solidary = parseSolidaryFrontmatter((solidaryContent as MarkdownModule).frontmatter);
const siteDescription = solidary.description || "Solidary Site Description.";

export async function getStaticPaths() {
  const pages = await getCollection("pages");
  return pages.map((entry) => ({
    params: { slug: entry.slug },
    props: { entry }
  }));
}

const { entry } = Astro.props;
const { Content } = await entry.render();
const title = entry.data.title;
const description = entry.data.description ?? siteDescription;
const canonicalPath = `/${entry.slug}`;
---

<Base title={title} description={description} canonicalPath={canonicalPath}>
  <article class="prose">
    <Content />
  </article>
</Base>
